cmake_minimum_required(VERSION 3.20)
project(JSDShowcase VERSION 0.4.2 LANGUAGES C CXX)
set(JSD_GAME_ENGINE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../jsd-game-engine" CACHE PATH "Sibling engine directory or installed SDK")
# Direct CMake works in a fresh source-only workspace without an installed SDK.
set(JSD_GAME_SOURCE_DEFAULT ON)
if(EXISTS "${JSD_GAME_ENGINE_DIR}/lib/cmake/JSDGameEngine/JSDGameEngineConfig.cmake")
    set(JSD_GAME_SOURCE_DEFAULT OFF)
endif()
option(JSD_GAME_USE_ENGINE_SOURCE "Build sibling engine from source" ${JSD_GAME_SOURCE_DEFAULT})
if(JSD_GAME_USE_ENGINE_SOURCE)
    set(JSD_BUILD_TESTS OFF CACHE BOOL "" FORCE)
    set(JSD_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
    add_subdirectory("${JSD_GAME_ENGINE_DIR}" engine-build EXCLUDE_FROM_ALL)
else()
    find_package(JSDGameEngine 0.4 CONFIG REQUIRED PATHS "${JSD_GAME_ENGINE_DIR}" NO_DEFAULT_PATH)
endif()
add_executable(sample-game src/main.cpp src/showcase.cpp)
option(JSD_GAME_STATIC_ENGINE "Link the static engine" OFF)
if(JSD_GAME_STATIC_ENGINE)
 target_link_libraries(sample-game PRIVATE JSD::GameEngineStatic)
else()
 target_link_libraries(sample-game PRIVATE JSD::GameEngine)
endif()
set(JSD_GAME_RUNTIME_ENGINE_DIR "${JSD_GAME_ENGINE_DIR}" CACHE PATH "Source-built SDK location at runtime")
file(RELATIVE_PATH JSD_GAME_ENGINE_RELATIVE_LIB "${CMAKE_CURRENT_SOURCE_DIR}/bin" "${JSD_GAME_RUNTIME_ENGINE_DIR}/lib")
set_target_properties(sample-game PROPERTIES INSTALL_RPATH "$ORIGIN/${JSD_GAME_ENGINE_RELATIVE_LIB}")
install(TARGETS sample-game RUNTIME DESTINATION bin)
