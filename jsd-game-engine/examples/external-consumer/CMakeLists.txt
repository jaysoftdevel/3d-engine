cmake_minimum_required(VERSION 3.20)
project(ExternalFrontierConsumer LANGUAGES CXX)
find_package(FrontierEngine 0.2 CONFIG REQUIRED)
add_executable(external-consumer main.cpp)
option(FRONTIER_CONSUMER_STATIC "Test static SDK target" OFF)
if(FRONTIER_CONSUMER_STATIC)
    target_link_libraries(external-consumer PRIVATE Frontier::EngineStatic)
else()
    target_link_libraries(external-consumer PRIVATE Frontier::Engine)
endif()
