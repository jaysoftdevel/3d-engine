cmake_minimum_required(VERSION 3.21)
include(cmake/ProjectGuards.cmake)
jsd_guard_project(jsd.engine 0.5.21)
project(JSDGameEngine VERSION 0.5.21 LANGUAGES C CXX)
set(JSD_ENGINE_STANDALONE OFF)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
 set(JSD_ENGINE_STANDALONE ON)
endif()
if(WIN32)
 if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Windows engine builds require a 64-bit toolchain")
 endif()
 set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
 set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
 add_compile_definitions(NOMINMAX WIN32_LEAN_AND_MEAN SDL_MAIN_HANDLED)
endif()
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
find_package(Threads REQUIRED)
include(cmake/JSDDependencies.cmake)
set(JSD_SOURCES src/shader_cache.cpp src/lines.cpp src/settings.cpp src/optics.cpp src/platform.cpp src/geometry.cpp src/renderer.cpp src/pipeline.cpp src/terrain.cpp src/ui.cpp src/advanced.cpp src/ibl.cpp src/local_lights.cpp src/lighting.cpp src/local_shadows.cpp src/lod.cpp src/scatter.cpp src/runtime/runtime.cpp src/runtime/assets.cpp src/runtime/project.cpp src/runtime/project_mesh.cpp src/runtime/project_io.cpp src/runtime/project_simulation.cpp src/runtime/project_terrain.cpp src/runtime/project_runtime.cpp src/runtime/project_capture.cpp src/runtime/physics.cpp src/runtime/vehicle.cpp src/runtime/strands.cpp src/runtime/audio.cpp src/cinematic.cpp src/mesh_tools.cpp src/runtime/particles.cpp src/runtime/cinematic.cpp src/runtime/camera_collision.cpp)
# Compile the engine once; both SDK libraries reuse the same PIC objects.
add_library(jsd-engine-objects OBJECT ${JSD_SOURCES})
target_compile_features(jsd-engine-objects PRIVATE cxx_std_20)
target_include_directories(jsd-engine-objects BEFORE PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/third_party/JoltPhysics")
target_include_directories(jsd-engine-objects PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_link_libraries(jsd-engine-objects PRIVATE Jolt SDL2::SDL2 Threads::Threads)
if(MSVC)
 target_compile_options(jsd-engine-objects PRIVATE /W4 /bigobj /utf-8 /permissive- /EHsc /Zc:__cplusplus)
else()
 target_compile_options(jsd-engine-objects PRIVATE -Wall -Wextra -Wpedantic -Wno-misleading-indentation)
endif()
set_target_properties(jsd-engine-objects PROPERTIES POSITION_INDEPENDENT_CODE ON)
add_library(jsd-game-engine SHARED $<TARGET_OBJECTS:jsd-engine-objects>)
add_library(jsd-game-engine-static STATIC $<TARGET_OBJECTS:jsd-engine-objects>)
if(WIN32)
 set_target_properties(jsd-game-engine PROPERTIES WINDOWS_EXPORT_ALL_SYMBOLS ON PREFIX "")
 set_target_properties(jsd-game-engine-static PROPERTIES OUTPUT_NAME jsd-game-engine-static)
endif()
add_library(JSD::GameEngine ALIAS jsd-game-engine)
add_library(JSD::GameEngineStatic ALIAS jsd-game-engine-static)
add_library(Frontier::Engine ALIAS jsd-game-engine)
add_library(Frontier::EngineStatic ALIAS jsd-game-engine-static)
add_library(frontier-engine ALIAS jsd-game-engine)
add_library(frontier-engine-static ALIAS jsd-game-engine-static)
set_target_properties(jsd-game-engine PROPERTIES VERSION ${PROJECT_VERSION} SOVERSION 0 EXPORT_NAME GameEngine INSTALL_RPATH "$ORIGIN")
set_target_properties(jsd-game-engine-static PROPERTIES OUTPUT_NAME "$<IF:$<BOOL:${WIN32}>,jsd-game-engine-static,jsd-game-engine>" EXPORT_NAME GameEngineStatic)
foreach(target jsd-game-engine jsd-game-engine-static)
 target_compile_features(${target} PUBLIC cxx_std_20)
 if(WIN32)
  target_compile_definitions(${target} INTERFACE NOMINMAX WIN32_LEAN_AND_MEAN SDL_MAIN_HANDLED)
 endif()
 target_include_directories(${target} BEFORE PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/third_party/JoltPhysics")
 set_target_properties(${target} PROPERTIES POSITION_INDEPENDENT_CODE ON)
 target_include_directories(${target} PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
 target_link_libraries(${target} PUBLIC Threads::Threads ${CMAKE_DL_LIBS} SDL2::SDL2 PRIVATE Jolt)
 if(MSVC)
  target_compile_options(${target} INTERFACE /utf-8 /Zc:__cplusplus)
 endif()
endforeach()
if(DEFINED FRONTIER_BUILD_EXAMPLES AND NOT DEFINED JSD_BUILD_EXAMPLES)
 set(JSD_BUILD_EXAMPLES "${FRONTIER_BUILD_EXAMPLES}" CACHE BOOL "Graphics laboratory")
endif()
if(DEFINED FRONTIER_BUILD_TESTS AND NOT DEFINED JSD_BUILD_TESTS)
 set(JSD_BUILD_TESTS "${FRONTIER_BUILD_TESTS}" CACHE BOOL "Engine regression suite")
endif()
option(JSD_BUILD_EXAMPLES "Graphics laboratory" ON)
option(JSD_BUILD_TESTS "Engine regression suite" ON)
option(JSD_BUILD_PROJECT_PLAYER "Build the standalone engine project player" ON)
if(JSD_BUILD_PROJECT_PLAYER)
 add_executable(jsd-project-player examples/project_player.cpp)
 target_link_libraries(jsd-project-player PRIVATE JSD::GameEngine)
 set_target_properties(jsd-project-player PROPERTIES INSTALL_RPATH "$ORIGIN/../${CMAKE_INSTALL_LIBDIR}")
if(JSD_ENGINE_STANDALONE)
 install(TARGETS jsd-project-player RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
endif()
endif()
option(JSD_BUILD_SHOWCASE "Build the optional SUNDOG autoplay showcase" OFF)
if(JSD_BUILD_EXAMPLES)
 add_executable(jsd-graphics-lab examples/demo.cpp)
 target_link_libraries(jsd-graphics-lab PRIVATE JSD::GameEngine)
 set_target_properties(jsd-graphics-lab PROPERTIES INSTALL_RPATH "$ORIGIN/../${CMAKE_INSTALL_LIBDIR}")
if(JSD_ENGINE_STANDALONE)
 install(TARGETS jsd-graphics-lab RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
endif()
endif()
if(JSD_BUILD_TESTS)
 enable_testing()
 add_executable(jsd-shader-data-tests tests/shader_data_tests.cpp)
 target_compile_features(jsd-shader-data-tests PRIVATE cxx_std_20)
 if(MSVC)
  target_compile_options(jsd-shader-data-tests PRIVATE /utf-8)
 endif()
 add_test(NAME shader-data COMMAND jsd-shader-data-tests "${CMAKE_CURRENT_SOURCE_DIR}/third_party/smaa/SMAA.hlsl")
 if(NOT WIN32)
 add_test(NAME version-guards COMMAND "${CMAKE_COMMAND}" -E env "CMAKE=${CMAKE_COMMAND}" "${JSD_GUARD_PYTHON}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_version_guards.py")
 endif()
 add_test(NAME windows-build-driver COMMAND "${JSD_GUARD_PYTHON}" -B "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_windows_build.py")
 if(UNIX)
  add_test(NAME shell-launchers COMMAND "${JSD_GUARD_PYTHON}" -B "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_shell_launchers.py")
 endif()
 add_test(NAME portable-guards COMMAND "${JSD_GUARD_PYTHON}" -B "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_portable_guards.py")
 add_test(NAME build-entrypoint-versions COMMAND "${JSD_GUARD_PYTHON}" -B "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_build_entrypoints.py")
 add_executable(jsd-project-tests tests/project_tests.cpp)
 target_link_libraries(jsd-project-tests PRIVATE JSD::GameEngine)
 add_test(NAME engine-projects COMMAND jsd-project-tests)
 if(JSD_BUILD_PROJECT_PLAYER)
  add_test(NAME project-cave-smoke COMMAND jsd-project-player --project "${CMAKE_CURRENT_SOURCE_DIR}/projects/sundog/sundog.jsdproject" --offscreen --frames 2 --time 116 --width 640 --height 360 --validate)
  add_test(NAME project-player-smoke COMMAND jsd-project-player --project "${CMAKE_CURRENT_SOURCE_DIR}/projects/sundog/sundog.jsdproject" --offscreen --frames 2 --time 40 --width 640 --height 360 --validate)
 endif()
 add_executable(jsd-engine-tests tests/regression.cpp)
 target_compile_definitions(jsd-engine-tests PRIVATE JSD_TEST_FIXTURE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures")
if(JSD_ENGINE_STANDALONE)
 install(DIRECTORY tests/fixtures/ DESTINATION ${CMAKE_INSTALL_DATADIR}/jsd-game-engine/test-fixtures)
endif()
 target_link_libraries(jsd-engine-tests PRIVATE JSD::GameEngine)
 set_target_properties(jsd-engine-tests PROPERTIES INSTALL_RPATH "$ORIGIN/../${CMAKE_INSTALL_LIBDIR}")
 add_executable(jsd-physics-tests tests/physics_tests.cpp)
 target_link_libraries(jsd-physics-tests PRIVATE JSD::GameEngine)
 set_target_properties(jsd-physics-tests PROPERTIES INSTALL_RPATH "$ORIGIN/../${CMAKE_INSTALL_LIBDIR}")
if(JSD_ENGINE_STANDALONE)
 install(TARGETS jsd-physics-tests RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
endif()
 add_test(NAME physics COMMAND jsd-physics-tests)
 add_test(NAME runtime-and-settings COMMAND jsd-engine-tests)
 add_test(NAME graphics COMMAND jsd-engine-tests --graphics)
if(JSD_ENGINE_STANDALONE)
 install(TARGETS jsd-engine-tests RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
endif()
endif()
if(JSD_BUILD_SHOWCASE)
 add_subdirectory(examples/showcase)
endif()
if(JSD_ENGINE_STANDALONE)
install(TARGETS jsd-game-engine jsd-game-engine-static Jolt EXPORT JSDTargets RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR} LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR} ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(DIRECTORY third_party/JoltPhysics/Jolt DESTINATION ${CMAKE_INSTALL_INCLUDEDIR} FILES_MATCHING PATTERN "*.h" PATTERN "*.inl")
install(DIRECTORY compat/red_frontier_0_1/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/frontier-compat/red_frontier_0_1)
install(EXPORT JSDTargets NAMESPACE JSD:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/JSDGameEngine)
configure_package_config_file(cmake/JSDGameEngineConfig.cmake.in "${CMAKE_CURRENT_BINARY_DIR}/JSDGameEngineConfig.cmake" INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/JSDGameEngine)
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/JSDGameEngineConfigVersion.cmake" VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/JSDGameEngineConfig.cmake" "${CMAKE_CURRENT_BINARY_DIR}/JSDGameEngineConfigVersion.cmake" DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/JSDGameEngine)

# Keep v0.2 source consumers working after rebuilding their application.
configure_package_config_file(cmake/FrontierEngineConfig.cmake.in "${CMAKE_CURRENT_BINARY_DIR}/FrontierEngineConfig.cmake" INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/FrontierEngine)
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/FrontierEngineConfigVersion.cmake" VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/FrontierEngineConfig.cmake" "${CMAKE_CURRENT_BINARY_DIR}/FrontierEngineConfigVersion.cmake" DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/FrontierEngine)
configure_file(cmake/jsd-game-engine.pc.in "${CMAKE_CURRENT_BINARY_DIR}/jsd-game-engine.pc" @ONLY)
configure_file(cmake/frontier-engine.pc.in "${CMAKE_CURRENT_BINARY_DIR}/frontier-engine.pc" @ONLY)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/jsd-game-engine.pc" "${CMAKE_CURRENT_BINARY_DIR}/frontier-engine.pc" DESTINATION ${CMAKE_INSTALL_LIBDIR}/pkgconfig)
if(UNIX AND NOT APPLE)
install(CODE "file(CREATE_LINK libjsd-game-engine.so \"\${CMAKE_INSTALL_PREFIX}/${CMAKE_INSTALL_LIBDIR}/libfrontier-engine.so\" SYMBOLIC)")
install(CODE "file(CREATE_LINK libjsd-game-engine.a \"\${CMAKE_INSTALL_PREFIX}/${CMAKE_INSTALL_LIBDIR}/libfrontier-engine.a\" SYMBOLIC)")
endif()
install(DIRECTORY projects DESTINATION ${CMAKE_INSTALL_DATADIR}/jsd-game-engine)
install(DIRECTORY docs licenses DESTINATION ${CMAKE_INSTALL_DATADIR}/jsd-game-engine)

install(FILES third_party/SDL2/LICENSE.txt DESTINATION ${CMAKE_INSTALL_DATADIR}/jsd-game-engine/licenses RENAME SDL2-LICENSE.txt)
jsd_install_receipt(jsd.engine 0.5.21)
endif()
